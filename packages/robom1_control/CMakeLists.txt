cmake_minimum_required(VERSION 3.8)
project(robom1_control)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
foreach(node base_sim goal_follower)
  add_executable(${node} src/${node}.cpp)
  target_include_directories(${node} PRIVATE include)
  ament_target_dependencies(${node} rclcpp geometry_msgs nav_msgs sensor_msgs tf2_ros)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_kinematics test/test_kinematics.cpp)
  target_include_directories(test_kinematics PRIVATE include)
endif()
ament_package()
